cmake_minimum_required(VERSION 3.18 FATAL_ERROR)

project(houghLines LANGUAGES CXX CUDA)

find_package(CUDAToolkit REQUIRED)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_executable(houghLines houghLines.cpp)
set_source_files_properties(houghLines.cpp PROPERTIES LANGUAGE CUDA)
target_link_libraries(houghLines PRIVATE CUDA::cudart CUDA::nppc CUDA::nppif)

install(TARGETS houghLines DESTINATION bin)
